% Pandoc maps Markdown headings to LaTeX commands as follows for the article
% class: H1 -> \section, H2 -> \subsection, H3 -> \subsubsection. The CV body
% starts with an H1 that duplicates the YAML-driven header, so \section is
% suppressed and the visible hierarchy is shifted one level down.

% H1 (suppressed: header is drawn from YAML metadata via \cvheader)
\titleformat{\section}{}{}{0pt}{}
\titlespacing*{\section}{0pt}{0pt}{0pt}
\renewcommand{\section}[1]{}

% H2 -> major section heading (accent colour + thin rule)
\titleformat{\subsection}
  {\large\sffamily\bfseries\color{accent}}
  {}
  {0pt}
  {}
  [\vspace{2pt}{\color{rulecolor}\titlerule[0.4pt]}]
\titlespacing*{\subsection}{0pt}{8pt}{5pt}

% H3 -> role / project heading (bold body colour, tight)
\titleformat{\subsubsection}
  {\normalsize\sffamily\bfseries\color{textbody}}
  {}
  {0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{6pt}{2pt}

% Header
\newcommand{\cvheader}[6]{%
  \begin{center}
    {\fontsize{26}{30}\selectfont\sffamily\bfseries\color{accent} #1\par}
    \vspace{2pt}
    {\fontsize{11}{13}\selectfont\sffamily\color{textbody} #2\par}
    \vspace{4pt}
    {\fontsize{9}{11}\selectfont\sffamily\color{muted}%
      #3%
      $if(email)$ \; \textbullet \; \href{mailto:#4}{#4}$endif$%
      $if(linkedin)$ \; \textbullet \; \href{#6}{LinkedIn}$endif$%
      $if(website)$ \; \textbullet \; \href{$if(website_url)$$website_url$$else$https://#5$endif$}{Website}$endif$%
      \par}
  \end{center}
  \vspace{4pt}
  {\color{rulecolor}\hrule height 0.4pt}
  \vspace{2pt}
}

  \vspace{4pt}
